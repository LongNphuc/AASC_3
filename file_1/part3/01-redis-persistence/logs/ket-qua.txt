== 2. Cấu hình bảo mật và bền vững dữ liệu trong /etc/redis/redis.conf
$ sudo grep -E '^(bind|protected-mode|port|requirepass|save|appendonly|appendfsync|aof-use-rdb-preamble) ' /etc/redis/redis.conf | mask
bind 127.0.0.1 -::1 172.31.0.164
protected-mode yes
port 6379
appendonly yes
appendfsync everysec
aof-use-rdb-preamble yes
requirepass <REDACTED>
save 300 10 60 1000

== 8. Mô phỏng sập đột ngột (kill -9) để kiểm tra AOF
$ sudo kill -9 "$(pidof redis-server)"
$ sleep 2; systemctl is-active redis || true
failed
$ redis-cli GET crash:test
ghi truoc khi kill -9
$ show_data | diff /tmp/aasc3-before.txt - && echo "=> Dữ liệu mẫu vẫn còn nguyên sau kill -9"
